\chapter{Thu thập \& Phân tích yêu cầu (Requirement Elicitation)}

\section{Các hệ thống liên quan (Related Works)}

\subsection{Nền tảng social commerce}

\subsubsection{TikTok Shop (mobile-first)}
\paragraph{Chức năng – TikTok Shop}
\begin{itemize}
    \item \textbf{Tài khoản \& truy cập:} Đăng ký/đăng nhập, khôi phục mật khẩu.
    \item \textbf{Nội dung \& tương tác:} Lướt feed video ngắn, like/comment/share, theo dõi creator.
    \item \textbf{Mua trong luồng:} Xem sản phẩm gắn trong nội dung, thêm giỏ, mua ngay.
    \item \textbf{Bán hàng live \& ưu đãi:} Livestream bán hàng, flash sale, voucher, freeship theo điều kiện.
    \item \textbf{Hậu mãi:} Theo dõi đơn, đánh giá sau mua, hỗ trợ đổi trả.
\end{itemize}

\paragraph{Thế mạnh – TikTok Shop}
\begin{itemize}
    \item \textbf{Nội dung ngắn \& chuyển đổi:} Nội dung cuốn hút, dẫn dắt nhu cầu mua sắm và tăng tỷ lệ chuyển đổi.
    \item \textbf{Mua trong luồng \& thanh toán:} In-stream purchase ít bước, hỗ trợ đa phương thức.
    \item \textbf{Cá nhân hóa:} Thuật toán đề xuất mạnh dựa trên hành vi xem và tương tác.
    \item \textbf{Công cụ sáng tạo:} Hiệu ứng, nhạc, sticker, template kịch bản cho seller/creator.
\end{itemize}

\paragraph{Hạn chế – TikTok Shop}
\begin{itemize}
    \item \textbf{Thiên di động:} Trải nghiệm trên trình duyệt/PC còn hạn chế.
    \item \textbf{Duyệt/so sánh trên web:} Khả năng điều hướng, so sánh sản phẩm kém hơn web thương mại điện tử truyền thống.
    \item \textbf{Quản trị \& phân tích:} UI quản trị desktop chưa trực quan cho seller nhỏ cần quản lý sản phẩm, đơn hàng và nội dung trong cùng một nơi.
\end{itemize}

\subsubsection{Facebook Marketplace \& Shops}
\paragraph{Chức năng – Facebook Marketplace/Shops}
\begin{itemize}
    \item \textbf{Rao bán \& tìm kiếm/lọc:} Đăng/duyệt tin, tìm kiếm, lọc theo vị trí, giá, danh mục.
    \item \textbf{Nhắn tin \& giao dịch:} Chat Messenger, thương lượng, hẹn giao dịch.
    \item \textbf{Catalog \& gắn sản phẩm:} Tạo catalog, trang cửa hàng, gắn sản phẩm lên bài/Story/Ads (Shops).
    \item \textbf{Checkout \& theo dõi:} Checkout (tùy thị trường), theo dõi đơn, phản hồi.
\end{itemize}

\paragraph{Thế mạnh – Facebook Marketplace/Shops}
\begin{itemize}
    \item \textbf{Niềm tin xã hội:} Hồ sơ cá nhân, bạn bè chung, cộng đồng lớn.
    \item \textbf{Quảng cáo \& remarketing:} Nhắm mục tiêu đa dạng, tích hợp Ads.
    \item \textbf{Đa nền tảng:} Mobile, web, desktop với UI quen thuộc.
\end{itemize}

\paragraph{Hạn chế – Facebook Marketplace/Shops}
\begin{itemize}
    \item \textbf{Mua trong luồng:} Chưa liền mạch ở nhiều thị trường và phụ thuộc chính sách từng khu vực.
    \item \textbf{Khả năng quản lý seller:} Công cụ quản lý sản phẩm, đơn hàng và báo cáo thương mại chưa tập trung như các nền tảng e-commerce chuyên biệt.
    \item \textbf{Live shopping:} Gắn sản phẩm vào livestream còn hạn chế so với các nền tảng social commerce chuyên về nội dung ngắn.
\end{itemize}

\paragraph{Kết luận về ưu và nhược điểm}
TikTok Shop mạnh về nội dung ngắn và mua trong luồng trên mobile nhưng chưa tối ưu cho thao tác quản trị và so sánh sản phẩm trên web/PC. Facebook Marketplace/Shops mạnh về niềm tin xã hội, cộng đồng và đa thiết bị nhưng trải nghiệm thương mại trong luồng còn phụ thuộc khu vực. Cơ hội của hệ thống là kết hợp feed xã hội, marketplace, hồ sơ người bán và các thao tác mua sắm cơ bản trong một nền tảng web dễ mở rộng.

\subsection{Phần mềm quản trị dịch vụ/bán hàng liên quan}

\subsubsection{Shopify (Admin \& POS)}
\paragraph{Chức năng – Shopify}
\begin{itemize}
    \item \textbf{Catalog \& tồn kho:} Quản lý sản phẩm, biến thể, giá, mã giảm giá, tồn kho.
    \item \textbf{Đơn hàng \& thanh toán:} Xử lý đơn, vận chuyển, hoàn tiền; hỗ trợ nhiều cổng thanh toán.
    \item \textbf{Báo cáo \& mở rộng:} Báo cáo bán hàng/kênh/chiến dịch; app store mở rộng chức năng.
    \item \textbf{Omnichannel:} Online store, social channel, POS.
\end{itemize}

\paragraph{Thế mạnh – Shopify}
\begin{itemize}
    \item \textbf{SaaS \& hệ sinh thái:} Ổn định, dễ khởi tạo, nhiều app mở rộng.
    \item \textbf{Báo cáo vận hành:} Doanh thu, đơn hàng, tồn kho và kênh bán được trình bày rõ ràng.
    \item \textbf{UI quản trị:} Web/desktop tốt, quy trình fulfillment rõ.
\end{itemize}

\paragraph{Hạn chế – Shopify}
\begin{itemize}
    \item \textbf{Chi phí:} Phí nền tảng/giao dịch cao với seller nhỏ.
    \item \textbf{Tùy biến sâu:} Cần app trả phí hoặc kỹ thuật riêng.
    \item \textbf{Social commerce:} Chưa phải trải nghiệm mạng xã hội gốc, việc kéo người dùng từ nội dung sang mua hàng thường phụ thuộc kênh bên ngoài.
\end{itemize}

\subsubsection{Meta Business Suite}
\paragraph{Chức năng – Meta Business Suite}
\begin{itemize}
    \item \textbf{Nội dung \& lịch đăng:} Quản lý Page, lịch đăng đa kênh (Facebook/Instagram).
    \item \textbf{Tương tác \& Inbox:} Hộp thư hợp nhất, nhãn hội thoại, auto-reply cơ bản, quản lý bình luận/tin nhắn.
    \item \textbf{Catalog \& Ads:} Quản lý catalog, gắn sản phẩm vào bài/Ads; báo cáo tương tác, hiệu quả quảng cáo.
\end{itemize}

\paragraph{Thế mạnh – Meta Business Suite}
\begin{itemize}
    \item \textbf{Hợp nhất trên web/desktop:} Quản trị nội dung và tương tác trong một nơi.
    \item \textbf{Liên kết Ads:} Hỗ trợ remarketing, đo lường chiến dịch.
    \item \textbf{UI thân thuộc:} Đa nền tảng, dễ làm quen.
\end{itemize}

\paragraph{Hạn chế – Meta Business Suite}
\begin{itemize}
    \item \textbf{Thương mại \& fulfillment:} Quản lý đơn/hoàn tiền chưa sâu như Shopify.
    \item \textbf{Mua trong luồng:} Trải nghiệm từ bài đăng sang đặt hàng chưa đồng nhất giữa các thị trường.
    \item \textbf{Workflow/automation:} Hạn chế, phụ thuộc API/third-party.
\end{itemize}

\paragraph{Ưu và nhược điểm}
Shopify mạnh về vận hành bán hàng và quản lý đơn hàng, trong khi Meta Business Suite mạnh về quản trị nội dung/xã hội trên web. Tuy nhiên, cả hai hướng tiếp cận này đều chưa hợp nhất đầy đủ trải nghiệm feed xã hội, marketplace, nhắn tin, quản lý seller và xử lý báo cáo vi phạm trong một hệ thống web thống nhất dành cho social commerce.

\paragraph{Kết luận}
Điểm mạnh cần kế thừa là niềm tin xã hội, trải nghiệm mua trong luồng, hỗ trợ đa thiết bị, quản lý sản phẩm/đơn hàng rõ ràng và báo cáo dễ hiểu. Khoảng trống cần giải quyết là thiếu sự hợp nhất giữa nội dung xã hội, thương mại, nhắn tin, cộng đồng, seller dashboard, công cụ AI hỗ trợ tạo nội dung và hệ thống admin kiểm duyệt trong cùng một nền tảng.

\section{Các bên liên quan (Stakeholders)}
\begin{itemize}
    \item \textbf{Guest (khách vãng lai):} Người dùng chưa đăng ký, có thể duyệt feed, xem bài, xem sản phẩm gắn trong bài và xem chi tiết sản phẩm. Guest \textbf{không} được phép thực hiện các tương tác xã hội (như like, comment, follow) và \textbf{không} thực hiện được các thao tác thương mại (như thêm vào giỏ hàng, đặt hàng, đánh giá).
    \item \textbf{User/Buyer đã đăng ký:} Người dùng đã có tài khoản, có thể theo dõi creator/seller, thêm sản phẩm vào giỏ, đặt hàng, xem lịch sử đơn, đánh giá sản phẩm sau mua, bình luận bài viết, lưu danh sách yêu thích, báo cáo vi phạm và nhận thông báo real-time.
    \item \textbf{Seller (đã được duyệt):} Người dùng nâng cấp vai trò thông qua quy trình phê duyệt; có thể tạo sản phẩm, gắn một hoặc nhiều sản phẩm vào bài viết, quản lý đơn/hoàn tiền, theo dõi đánh giá, xem dashboard hiệu suất, lên lịch đăng và dùng AI hỗ trợ tạo nội dung/caption.
    \item \textbf{Admin/Moderator:} Quản trị hệ thống qua ứng dụng admin riêng; có thể phê duyệt seller, quản lý người dùng, quản lý danh mục, kiểm duyệt bài viết/sản phẩm, xử lý báo cáo vi phạm và xem dashboard tổng hợp.
\end{itemize}

\section{Yêu cầu chức năng và phi chức năng}

\subsection{Yêu cầu chức năng}

\subsubsection{Chức năng chung cho Guest}
\begin{itemize}
    \item Đăng ký tài khoản, đăng nhập, xác thực email và khôi phục mật khẩu.
    \item Xem feed nội dung công khai, xem bài viết, tìm kiếm nội dung và sản phẩm theo từ khóa.
    \item Duyệt marketplace; lọc/sắp xếp sản phẩm theo danh mục, khoảng giá, đánh giá hoặc tiêu chí sắp xếp được hỗ trợ.
    \item Xem chi tiết sản phẩm: giá, biến thể, tồn kho, mô tả, media, đánh giá và thông tin người bán.
    \item Xem hồ sơ công khai của buyer/seller và nội dung công khai liên quan.
    \item Chỉ được sử dụng các thao tác cá nhân hóa và thương mại đầy đủ sau khi đăng ký/đăng nhập.
\end{itemize}

\subsubsection{Chức năng cho User/Buyer đã đăng ký}
\begin{itemize}
    \item Quản lý hồ sơ: cập nhật thông tin, đổi mật khẩu, bật/tắt 2FA và cấu hình quyền riêng tư cơ bản.
    \item Theo dõi/huỷ theo dõi seller/creator; lưu bài viết hoặc sản phẩm vào danh sách yêu thích.
    \item Quản lý giỏ hàng: thêm/xóa/cập nhật số lượng sản phẩm và chọn thông tin giao hàng khi đặt hàng.
    \item Đặt hàng, xác nhận thanh toán theo phương thức được hỗ trợ và theo dõi trạng thái đơn với thông báo real-time.
    \item Đánh giá sản phẩm sau mua, bình luận bài viết và báo cáo bài viết, bình luận, sản phẩm hoặc người dùng vi phạm.
    \item Nhắn tin trực tiếp với người dùng khác và nhận thông báo real-time về đơn hàng, tương tác xã hội, tin nhắn và hệ thống.
    \item Nhận gợi ý sản phẩm/nội dung dựa trên hành vi xem, tìm kiếm và tương tác trong hệ thống; sử dụng AI Shopping Assistant để hỏi giá, tồn kho, so sánh sản phẩm và tra cứu đơn hàng của bản thân.
\end{itemize}

\subsubsection{Chức năng cho Seller}
\begin{itemize}
    \item Nâng cấp tài khoản thành seller bằng cách gửi hồ sơ xác minh, thông tin kinh doanh/ngân hàng và chờ admin duyệt.
    \item Quản lý sản phẩm: tạo/sửa/xóa/khôi phục sản phẩm, quản lý hình ảnh, biến thể, giá, tồn kho và trạng thái hiển thị; gắn một hoặc nhiều sản phẩm vào bài viết; lên lịch đăng bài.
    \item Quản lý đơn hàng và hoàn tiền: xem, lọc, cập nhật trạng thái, xử lý yêu cầu hủy/hoàn tiền theo quy trình được hỗ trợ.
    \item Theo dõi đánh giá sản phẩm, phản hồi của khách hàng và các chỉ số vận hành trong seller dashboard.
    \item Sử dụng AI để tạo nội dung chữ, caption, hashtag, lời kêu gọi hành động và hình ảnh marketing khi nhà cung cấp AI được cấu hình.
    \item Dashboard: xem doanh thu, đơn hàng mới/đang chờ, lượt xem sản phẩm, tồn kho thấp và các biểu đồ/tổng quan hiệu suất bán hàng.
\end{itemize}

\subsubsection{Chức năng cho Admin/Moderator}
\begin{itemize}
    \item Quản trị người dùng và seller: xem danh sách, tìm kiếm/lọc, khóa/mở khóa tài khoản, thay đổi vai trò phù hợp và duyệt/từ chối hồ sơ seller.
    \item Kiểm duyệt nội dung: xem, lọc và xóa bài viết/sản phẩm không phù hợp; xử lý báo cáo vi phạm đối với bài viết, bình luận, sản phẩm và người dùng.
    \item Quản lý danh mục sản phẩm.
    \item Giám sát và báo cáo: xem dashboard hệ thống với tổng số user/seller/buyer, sản phẩm, bài viết, đơn hàng, người dùng mới, đơn hàng mới, doanh thu và tăng trưởng theo thời gian.
    \item Theo dõi thông tin audit liên quan đến hồ sơ seller và các thay đổi dữ liệu nhạy cảm trong phạm vi được hệ thống hỗ trợ.
\end{itemize}

\subsection{Yêu cầu phi chức năng}
\begin{itemize}
    \item \textbf{Usability:} Giao diện thân thiện, responsive cho laptop/tablet/mobile; các luồng chính như duyệt feed, tìm sản phẩm, đặt hàng, đăng bài và quản lý seller dashboard cần rõ ràng, dễ thao tác.
    \item \textbf{Security:} Mật khẩu được băm bằng bcrypt; mật khẩu tối thiểu 8 ký tự, có chữ hoa, chữ thường và số; hỗ trợ xác thực email, khôi phục mật khẩu và 2FA; phân quyền theo vai trò buyer/seller/admin; mã hóa dữ liệu nhạy cảm trong hồ sơ seller; ghi nhận audit cho các thao tác nhạy cảm liên quan đến xác minh seller.
    \item \textbf{Performance:} Thời gian phản hồi mục tiêu: tải trang 1–3 giây; tìm kiếm/lọc 1–2 giây; thanh toán 5–7 giây; đăng ký/đăng nhập 1–3 giây.
    \item \textbf{Scalability/Throughput:} Hỗ trợ tối thiểu 1.000 yêu cầu đồng thời mà không suy giảm đáng kể hiệu năng; có thể mở rộng theo chiều ngang cho dịch vụ nội dung và thông báo real-time.
    \item \textbf{Availability:} Hệ thống sẵn sàng 24/7; có cơ chế giám sát, cảnh báo; phương án phục hồi sự cố (backup/restore) cho CSDL và media.
\end{itemize}
